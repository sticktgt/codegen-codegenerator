from __future__ import annotations

from demo_app.domain.models import Ticket
from demo_app.services.notification_service import build_assignment_message


def test_build_assignment_message_success() -> None:
    ticket = Ticket(ticket_id='T-1', title='Broken button', description='Save button does not work')
    message = build_assignment_message(ticket, 'alice')
    assert message == 'Тикет T-1 назначен агенту alice.'
